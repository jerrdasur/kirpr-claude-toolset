# CRUD по одиночной задаче — план реализации

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Добавить создание, обновление и удаление одиночной задачи в YouTrack и Jira — два независимых Python-скрипта и три slash-команды поверх них.

**Architecture:** Два самодостаточных скрипта (`manage_youtrack.py`, `manage_jira.py`) без общего модуля — API трекеров расходятся слишком сильно. Каждый содержит одинаковый скелет: транспорт с инъекцией для тестов, класс клиента, чистые функции рендера, `run()` для оркестрации, `main()` для argparse. Три slash-команды — тонкие промпты, выбирающие скрипт по `tracker.md`.

**Tech Stack:** Python 3 (только stdlib: `argparse`, `difflib`, `json`, `urllib`, `base64`), `unittest` для тестов, Markdown для промптов команд, POSIX sh для установщика.

## Global Constraints

- **Только stdlib.** Никаких внешних зависимостей в Python-скриптах — это правило репозитория (`README.md`, раздел «Зависимости»).
- **Скрипты плоские.** Все файлы кладутся прямо в `kirpr/scripts/`, без подкаталогов: `install.sh` копирует `kirpr/scripts/*.py` глобом и подкаталог не увидит.
- **Никакого общего модуля** между `manage_youtrack.py` и `manage_jira.py`. Дублирование скелета — принятое решение спеки, не недосмотр. Не рефакторить в общий файл.
- **Jira REST v2**, не v3. Базовый путь `/rest/api/2/`. Описание — обычная строка, не ADF.
- **YouTrack REST**: базовый путь `/api/`.
- **Никаких правок** `create_youtrack.py`, `fetch_youtrack.py`, `fetch_jira.py`, `tasks-push.md`. Новые команды изолированы от декомпозиции.
- **Язык вывода** скриптов и промптов — русский, как в остальном репозитории. Docstrings и имена — английские, как в `create_youtrack.py`.
- **Переменные окружения:** YouTrack — `YOUTRACK_URL`, `YOUTRACK_TOKEN`. Jira — `JIRA_URL`, `JIRA_TOKEN`, `JIRA_EMAIL` (опционально; его наличие означает Jira Cloud).
- **Тесты запускаются** из корня репозитория: `python3 -m unittest discover -s tests -v`.

class FakeTransport:
    """Records calls and replays canned (status, payload) responses in order."""

    def __init__(self, responses=None):
        self.calls = []
        self._responses = list(responses or [])

    def __call__(self, method, url, headers, data):
        body = json.loads(data.decode("utf-8")) if data else None
        self.calls.append((method, url, body))
        if not self._responses:
            raise AssertionError(f"unexpected call: {method} {url}")
        status, payload = self._responses.pop(0)
        raw = b"" if payload is None else json.dumps(payload).encode("utf-8")
        return status, raw

    def methods(self):
        return [c[0] for c in self.calls]

#!/usr/bin/env python3
"""Create, update and delete a single YouTrack issue.

Usage:
  manage_youtrack.py [--dry-run] create --project SAS --summary "..."
  manage_youtrack.py [--dry-run] update SAS-201 --command "State In Progress"
  manage_youtrack.py [--dry-run] delete SAS-201 --confirm SAS-201

Requires YOUTRACK_URL and YOUTRACK_TOKEN environment variables.
"""

import argparse
import difflib
import json
import os
import sys
import urllib.error
import urllib.request


class ArgError(ValueError):
    """Raised when CLI arguments are missing or inconsistent."""


class ApiError(RuntimeError):
    """Raised when the YouTrack API rejects a request."""


def _urllib_transport(method, url, headers, data):
    req = urllib.request.Request(url, data=data, headers=headers, method=method)
    try:
        with urllib.request.urlopen(req, timeout=30) as resp:
            return resp.status, resp.read()
    except urllib.error.HTTPError as e:
        return e.code, e.read()


def explain_http(status: int, method: str, path: str, raw: bytes) -> str:
    text = raw.decode("utf-8", "replace") if raw else ""
    if status in (401, 403):
        hint = " Проверь YOUTRACK_TOKEN и права пользователя."
    elif status == 404:
        hint = " Задача или проект не найден."
    else:
        hint = ""
    return f"YouTrack API {method} {path} -> HTTP {status}: {text}{hint}"


class YouTrackClient:
    def __init__(self, base_url: str, token: str, transport=None):
        self.base_url = base_url.rstrip("/")
        self.token = token
        self._transport = transport or _urllib_transport

    def _request(self, method: str, path: str, body=None):
        url = f"{self.base_url}{path}"
        headers = {
            "Accept": "application/json",
            "Authorization": f"Bearer {self.token}",
            "Cache-Control": "no-cache",
            "Content-Type": "application/json",
        }
        data = json.dumps(body).encode("utf-8") if body is not None else None
        status, raw = self._transport(method, url, headers, data)
        if status >= 400:
            raise ApiError(explain_http(status, method, path, raw))
        if not raw:
            return None
        return json.loads(raw.decode("utf-8"))


def resolve_description(args):
    inline = getattr(args, "description", None)
    path = getattr(args, "description_file", None)
    if inline is not None and path is not None:
        raise ArgError("--description и --description-file взаимоисключающие.")
    if path is not None:
        try:
            with open(path, "r", encoding="utf-8") as f:
                return f.read()
        except OSError as e:
            raise ArgError(f"Не читается --description-file: {e}")
    return inline


def validate_update(args) -> None:
    changes = [
        args.summary,
        args.description,
        args.description_file,
        args.command,
        args.comment,
    ]
    if not any(c is not None and c != [] for c in changes):
        raise ArgError(
            "update требует хотя бы один из флагов: --summary, --description, "
            "--description-file, --command, --comment."
        )


def validate_delete(args) -> None:
    if not args.confirm:
        raise ArgError(
            f"delete требует --confirm {args.issue}. Удаление необратимо."
        )
    if args.confirm != args.issue:
        raise ArgError(
            f"--confirm {args.confirm!r} не совпадает с задачей {args.issue!r}."
        )


def render_description_diff(old, new):
    old_lines = (old or "").splitlines()
    new_lines = (new or "").splitlines()
    diff = difflib.ndiff(old_lines, new_lines)
    return [line for line in diff if not line.startswith("? ")]


DRY_RUN_FOOTER = "_dry-run: сетевых записей не выполнено._"


def _commands_section(commands):
    if not commands:
        return []
    lines = ["## Команды", ""]
    for query in commands:
        lines.append(f"- `{query}`")
    lines.append("")
    return lines


def _comment_section(comment):
    if not comment:
        return []
    return ["## Комментарий", "", comment, ""]


def render_dry_run_create(project, summary, description, parent, commands,
                          comment):
    lines = ["# Dry-run: создание задачи", ""]
    lines.append(f"**Проект:** {project}")
    lines.append(f"**Заголовок:** {summary}")
    if parent:
        lines.append(f"**Родитель:** subtask of {parent}")
    lines.append("")
    if description:
        lines.extend(["## Описание", "", description, ""])
    lines.extend(_commands_section(commands))
    lines.extend(_comment_section(comment))
    lines.append(DRY_RUN_FOOTER)
    return "\n".join(lines)


def render_dry_run_update(current, summary, description, commands, comment):
    issue_id = current.get("idReadable") or "?"
    lines = [f"# Dry-run: обновление {issue_id}", ""]
    lines.append(f"**Задача:** {current.get('summary') or '—'}")
    lines.append("")
    if summary:
        lines.append("| поле | было | станет |")
        lines.append("|------|------|--------|")
        lines.append(f"| summary | {current.get('summary') or '—'} | {summary} |")
        lines.append("")
    if description is not None:
        lines.append("## Описание")
        lines.append("")
        lines.extend(render_description_diff(current.get("description"),
                                             description))
        lines.append("")
    lines.extend(_commands_section(commands))
    lines.extend(_comment_section(comment))
    lines.append(DRY_RUN_FOOTER)
    return "\n".join(lines)


def render_delete_preview(preview):
    issue_id = preview.get("idReadable") or "?"
    reporter = preview.get("reporter") or {}
    author = reporter.get("name") or reporter.get("login") or "—"
    comments = preview.get("comments") or []
    links = preview.get("links") or []
    lines = [f"# Dry-run: удаление {issue_id}", ""]
    lines.append(f"**Задача:** {preview.get('summary') or '—'}")
    lines.append(f"**Автор:** {author}")
    lines.append(f"**Комментариев:** {len(comments)}")
    lines.append("")
    related = []
    for link in links:
        name = (link.get("linkType") or {}).get("name") or "link"
        for issue in link.get("issues") or []:
            if issue.get("idReadable"):
                related.append(f"- {name}: {issue['idReadable']}")
    if related:
        lines.append("## Связи")
        lines.append("")
        lines.extend(related)
        lines.append("")
    lines.append("**Удаление необратимо:** задача исчезнет вместе с "
                 "комментариями, связями и историей.")
    lines.append("")
    lines.append(DRY_RUN_FOOTER)
    return "\n".join(lines)


def render_report(title, applied, failed):
    lines = [f"# {title}", ""]
    if applied:
        lines.append("## Выполнено")
        lines.append("")
        for item in applied:
            lines.append(f"- {item}")
    else:
        lines.append("_ничего не выполнено_")
    if failed:
        lines.append("")
        lines.append("## Прервано")
        lines.append("")
        lines.append(failed)
        lines.append("")
        lines.append("_Откат не выполняется. Повторный запуск безопасен._")
    return "\n".join(lines)


def apply_steps(steps):
    applied = []
    for label, action in steps:
        try:
            action()
        except ApiError as e:
            return applied, f"{label}: {e}"
        applied.append(label)
    return applied, None


def build_parser():
    parser = argparse.ArgumentParser(
        description="Создание, обновление и удаление задачи в YouTrack."
    )
    parser.add_argument("--dry-run", action="store_true",
                        help="Предпросмотр; без сетевых записей.")
    sub = parser.add_subparsers(dest="action", required=True)

    create = sub.add_parser("create", help="Создать задачу.")
    create.add_argument("--project", required=True, help="Короткое имя проекта.")
    create.add_argument("--summary", required=True)
    create.add_argument("--description")
    create.add_argument("--description-file")
    create.add_argument("--parent", help="Задача-родитель (subtask of).")
    create.add_argument("--command", action="append", default=[],
                        help="Команда YouTrack, можно повторять.")
    create.add_argument("--comment")

    update = sub.add_parser("update", help="Обновить задачу.")
    update.add_argument("issue")
    update.add_argument("--summary")
    update.add_argument("--description")
    update.add_argument("--description-file")
    update.add_argument("--command", action="append", default=[])
    update.add_argument("--comment")

    delete = sub.add_parser("delete", help="Удалить задачу безвозвратно.")
    delete.add_argument("issue")
    delete.add_argument("--confirm", help="Номер задачи, введённый повторно.")

    return parser


def make_client():
    base_url = os.environ.get("YOUTRACK_URL")
    token = os.environ.get("YOUTRACK_TOKEN")
    if not base_url or not token:
        raise ArgError("YOUTRACK_URL и YOUTRACK_TOKEN должны быть заданы.")
    return YouTrackClient(base_url, token)


def run(args, client=None):
    raise NotImplementedError


def main(argv=None) -> int:
    args = build_parser().parse_args(argv)
    try:
        output, code = run(args)
    except (ArgError, ApiError) as e:
        print(f"Error: {e}", file=sys.stderr)
        return 1
    print(output)
    return code


if __name__ == "__main__":
    sys.exit(main())

class FakeTransport:
    """Records calls and replays canned (status, payload) responses in order."""

    def __init__(self, responses=None):
        self.calls = []
        self._responses = list(responses or [])

    def __call__(self, method, url, headers, data):
        body = json.loads(data.decode("utf-8")) if data else None
        self.calls.append((method, url, body, headers.get("Authorization")))
        if not self._responses:
            raise AssertionError(f"unexpected call: {method} {url}")
        status, payload = self._responses.pop(0)
        raw = b"" if payload is None else json.dumps(payload).encode("utf-8")
        return status, raw

    def methods(self):
        return [c[0] for c in self.calls]

def _urllib_transport(method, url, headers, data):
    req = urllib.request.Request(url, data=data, headers=headers, method=method)
    try:
        with urllib.request.urlopen(req, timeout=30) as resp:
            return resp.status, resp.read()
    except urllib.error.HTTPError as e:
        return e.code, e.read()


def explain_http(status: int, method: str, path: str, raw: bytes) -> str:
    text = raw.decode("utf-8", "replace") if raw else ""
    if status in (401, 403):
        hint = " Проверь JIRA_TOKEN, JIRA_EMAIL и права пользователя."
    elif status == 404:
        hint = " Задача или проект не найден."
    else:
        hint = ""
    return f"Jira API {method} {path} -> HTTP {status}: {text}{hint}"


class JiraClient:
    def __init__(self, base_url: str, token: str, email=None, transport=None):
        self.base_url = base_url.rstrip("/")
        self.token = token
        self.email = email or None
        self._transport = transport or _urllib_transport

    @property
    def is_cloud(self) -> bool:
        return bool(self.email)

    def _auth_header(self) -> str:
        if self.is_cloud:
            raw = f"{self.email}:{self.token}".encode("utf-8")
            return f"Basic {base64.b64encode(raw).decode('ascii')}"
        return f"Bearer {self.token}"

    def _request(self, method: str, path: str, body=None):
        url = f"{self.base_url}{path}"
        headers = {
            "Accept": "application/json",
            "Authorization": self._auth_header(),
            "Content-Type": "application/json",
        }
        data = json.dumps(body).encode("utf-8") if body is not None else None
        status, raw = self._transport(method, url, headers, data)
        if status >= 400:
            raise ApiError(explain_http(status, method, path, raw))
        if not raw:
            return None
        return json.loads(raw.decode("utf-8"))

def find_transition(transitions, target_status: str):
    wanted = (target_status or "").strip().lower()
    for transition in transitions:
        name = ((transition.get("to") or {}).get("name") or "").strip().lower()
        if name == wanted:
            return transition
    available = sorted({
        (t.get("to") or {}).get("name")
        for t in transitions
        if (t.get("to") or {}).get("name")
    })
    listed = ", ".join(available) if available else "—"
    raise ArgError(
        f"Из текущего статуса нет перехода в {target_status!r}. "
        f"Доступны: {listed}."
    )


def resolve_assignee(client, value: str):
    if value == UNASSIGN:
        return None
    if not client.is_cloud:
        return {"name": value}
    users = client.search_user(value)
    if not users:
        raise ArgError(f"Пользователь {value!r} не найден в Jira.")
    if len(users) > 1:
        names = ", ".join(
            u.get("displayName") or u.get("emailAddress") or u.get("accountId")
            or "?" for u in users
        )
        raise ArgError(
            f"Запрос {value!r} совпал с несколькими пользователями: {names}. "
            "Уточни запрос."
        )
    account_id = users[0].get("accountId")
    if not account_id:
        raise ArgError(f"У пользователя {value!r} нет accountId.")
    return {"accountId": account_id}

def _current_value(fields: dict, name: str) -> str:
    value = fields.get(name)
    if value is None:
        return "—"
    if isinstance(value, dict):
        return (value.get("name") or value.get("displayName")
                or value.get("key") or "—")
    if isinstance(value, list):
        return ", ".join(str(v) for v in value) if value else "—"
    return str(value)


def collect_changes(current, args, client):
    fields = current.get("fields") or {}
    changes = []
    if args.summary:
        changes.append(("summary", _current_value(fields, "summary"),
                        args.summary))
    if args.status is not None:
        find_transition(client.list_transitions(current.get("key")),

        changes.append(("assignee", _current_value(fields, "assignee"), after))

def _status_step(client, key, status):
    def action():
        transition = find_transition(client.list_transitions(key), status)
        client.do_transition(key, transition["id"])

    return (f"статус переведён в `{status}`", action)


def run_create(args, client, dry_run):
    description = resolve_description(args)
    if dry_run:
        # Ранняя проверка проекта: опечатка в --project должна вскрыться
        # до подтверждения, а не после.

        return render_dry_run_create(args, description), 0
    fields = build_fields(args, client, description, for_create=True)
    key = client.create_issue(fields)
    steps = []
    if args.status is not None:
        steps.append(_status_step(client, key, args.status))
    if args.comment:
        steps.append(("добавлен комментарий",
                      lambda: client.add_comment(key, args.comment)))
    applied, failed = apply_steps(steps)
    applied.insert(0, f"создана задача {key}")
    return render_report(f"Задача {key} создана", applied, failed), \
        (1 if failed else 0)


def run_update(args, client, dry_run):
    validate_update(args)
    description = resolve_description(args)
    current = client.fetch_issue(args.issue)
    current.setdefault("key", args.issue)
    if dry_run:
        changes = collect_changes(current, args, client)
        diff = None
        if description is not None:
            diff = render_description_diff(
                (current.get("fields") or {}).get("description"), description)
        return render_dry_run_update(current, changes, diff, args.comment), 0
    fields = build_fields(args, client, description, for_create=False)
    steps = []
    if fields:
        label = f"обновлены поля: {', '.join(sorted(fields))}"
        steps.append((label, lambda: client.update_issue(args.issue, fields)))
    if args.status is not None:
        steps.append(_status_step(client, args.issue, args.status))
    if args.comment:
        steps.append(("добавлен комментарий",
                      lambda: client.add_comment(args.issue, args.comment)))
    applied, failed = apply_steps(steps)
    return render_report(f"Задача {args.issue} обновлена", applied, failed), \
        (1 if failed else 0)


def run(args, client=None):
    if client is None:
        client = make_client()
    if args.action == "create":
        return run_create(args, client, args.dry_run)
    if args.action == "update":
        return run_update(args, client, args.dry_run)
    raise ArgError(f"Неизвестное действие: {args.action}")
